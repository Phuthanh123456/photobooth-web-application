cmake_minimum_required(VERSION 3.24)
project(photobooth_nikon_bridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Nikon bridge requires Windows x64")
endif()
if(NOT DEFINED ENV{NIKON_SDK_ROOT})
  message(FATAL_ERROR "Set NIKON_SDK_ROOT to the externally installed SDK")
endif()
add_executable(nikon-bridge bridge.cpp)
target_compile_features(nikon-bridge PRIVATE cxx_std_17)
target_include_directories(nikon-bridge PRIVATE "$ENV{NIKON_SDK_ROOT}/Module/Win/HeaderFile")
target_compile_definitions(nikon-bridge PRIVATE NOMINMAX CSLAYER)
target_link_libraries(nikon-bridge PRIVATE crypt32)
