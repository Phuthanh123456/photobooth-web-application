cmake_minimum_required(VERSION 3.24)
project(photobooth_nikon_bridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Nikon bridge requires Windows x64")
endif()
if(NOT DEFINED ENV{NIKON_SDK_ROOT})
  message(FATAL_ERROR "Set NIKON_SDK_ROOT to the externally installed SDK")
endif()
add_executable(nikon-bridge bridge.cpp)
add_executable(readiness-test tests/readiness.cpp)
foreach(target nikon-bridge readiness-test)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "$ENV{NIKON_SDK_ROOT}/Module/Win/HeaderFile")
  target_compile_definitions(${target} PRIVATE NOMINMAX CSLAYER)
  target_link_libraries(${target} PRIVATE crypt32)
endforeach()
